rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAI() {
      return request.auth != null && request.auth.token.role == 'ai';
    }
    
    function isNewUser() {
      // Allow operations when the document is being created for the first time
      return request.auth != null && 
             request.auth.uid == request.resource.data.uid &&
             resource == null;
    }

    // USERS
    match /users/{userId} {
      // Allow users to read and write their own data
      // Also allow creating new user profiles during registration
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
                   (request.auth.uid == userId || isNewUser());

      match /services/{serviceId} {
        allow read: if true;
        allow write: if request.auth != null &&
          (request.auth.uid == userId || isAI());
      }

      match /reviews/{reviewId} {
        allow read: if true;
        allow create: if request.auth != null || isAI();
        allow update, delete: if false;
      }

      match /notifications/{notificationId} {
        allow read, write: if request.auth != null &&
          (request.auth.uid == userId || isAI());
      }
    }

    // CHATS
    match /chats/{chatId} {
      allow read, write, create: if request.auth != null &&
        (
          (resource.data.participants != null && request.auth.uid in resource.data.participants) || 
          (request.resource.data.participants != null && request.auth.uid in request.resource.data.participants)
        );
    }

    // CHAT THREADS + MESSAGES
    match /chatThreads/{threadId} {
      allow read, write, create: if request.auth != null &&
        (
          (resource.data.participants != null && request.auth.uid in resource.data.participants) || 
          (request.resource.data.participants != null && request.auth.uid in request.resource.data.participants)
        );

      match /messages/{messageId} {
        allow read, write, create: if request.auth != null &&
          (
            (resource.data.senderId != null && request.auth.uid == resource.data.senderId) || 
            (request.resource.data.senderId != null && request.auth.uid == request.resource.data.senderId)
          );
      }
    }

    // APPOINTMENTS
    match /appointments/{appointmentId} {
      allow create: if request.auth != null;
      allow read, update, delete: if request.auth != null && 
        (request.auth.uid == resource.data.customerId || 
         request.auth.uid == resource.data.barberId);
    }

    // PAYMENTS
    match /payments/{paymentId} {
      allow read: if request.auth != null && 
                 (request.auth.uid == resource.data.customerId || 
                  request.auth.uid == resource.data.barberId);
      allow create: if request.auth != null;
    }

    match /appointmentPayments/{appointmentId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update: if request.auth != null && 
                    (request.auth.uid == resource.data.customerId || 
                     request.auth.uid == resource.data.barberId);
    }
    
    // INVOICES
    match /invoices/{invoiceId} {
      allow read: if request.auth != null && (
        resource.data.customerId == request.auth.uid ||
        resource.data.barberId == request.auth.uid
      );
      allow create: if request.auth != null && request.resource.data.customerId == request.auth.uid;
      allow update: if request.auth != null && (
        (request.auth.uid == resource.data.customerId && request.resource.data.status == 'paid') ||
        request.auth.uid == resource.data.barberId
      );
    }

    // BARBER AVAILABILITY
    match /availableSlots/{slotId} {
      allow read, write: if request.auth != null;
    }

    // BULLETIN BOARD
    match /bulletins/{postId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow delete: if request.auth != null &&
        (request.auth.uid == resource.data.authorId || isAI());

      match /comments/{commentId} {
        allow read: if true;
        allow create: if request.auth != null;
      }
    }

    // BARBERS (if separate collection)
    match /barbers/{barberId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == barberId;
    }

    // FOR PRODUCTION AND TESTFLIGHT: More restrictive than development
    match /{document=**} {
      // Only allow authenticated users to read any document
      allow read: if request.auth != null;
      
      // NO DEFAULT WRITE ACCESS
      // All write operations must be explicitly allowed by the rules above
    }
  }
}
